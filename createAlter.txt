CREATE TABLE personal(
      firstName VARCHAR2(10),
      lastName VARCHAR2(10),
      person_ID NUMBER(10),
      dob DATE,
      email VARCHAR2(15)
    ); 


 CREATE TABLE education(
    startYear DATE,
    endYear DATE,
    lastName VARCHAR2(10),
    institution VARCHAR2(10),
    degree VARCHAR2(10),
    contact NUMBER(10),
    person_ID NUMBER(10)
   );



 CREATE TABLE address(
    address_ID NUMBER(10),
    city VARCHAR2(10),
    country VARCHAR2(10),
    contact NUMBER(10),
    person_ID NUMBER(10),
    streetName VARCHAR2(10)
   );
